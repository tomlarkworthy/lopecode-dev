**The host: one runtime in a Durable Object, handler cells kept observed.** A variable defined with an observer stays computed; \`module.value()\` alone recomputes it every call.

\`\`\`js
import "./shim.js";
import { Runtime } from "./rt/index.js";          // @observablehq/runtime source files, unbundled
import define from "./m/@tom/a.js";               // the entry module, as the notebook holds it
import { DurableObject } from "cloudflare:workers";

export class State extends DurableObject {
  constructor(ctx, env) {
    super(ctx, env);
    this.runtime = new Runtime();
    this.main = this.runtime.module(define);
    this.main.variable({}).define("__handler", ["fetch"], (f) => f);   // pins the fetch cell
  }
  async fetch(request) { return (await this.main.value("fetch"))(request, this.env, this.ctx); }
  async alarm() { for (const job of await this.main.value("jobs")) await job.run(this.env); }
}

export default {
  fetch(request, env) {
    // redirect: "manual", or a 302 from a module is followed inside the Worker
    return env.STATE.get(env.STATE.idFromName("main")).fetch(request, { redirect: "manual" });
  },
};

**A module file with a relative import.** Bare specifiers are refused at upload.

\`\`\`js
// m/@tom/a.js
import define1 from "./b.js";
export default function define(runtime, observer) {
  const main = runtime.module();
  main.define("module @tom/b", async () => runtime.module(define1));
  main.define("greeting", ["module @tom/b", "@variable"], (_, v) => v.import("greeting", _));
  main.variable(observer("fetch")).define("fetch", ["greeting"], (greeting) => (request) => new Response(greeting));
  return main;
}

**The definition proxy.** A compiled cell's definition string becomes a native function the host runtime can hold. Functions cross in both directions.

\`\`\`js
const toGuest = (x) => {
  if (typeof x === "function") return vm.newFunction(x.name || "host", (...hs) => toGuest(x(...hs.map((h) => vm.dump(h)))));
  return vm.evalCode("(" + JSON.stringify(x ?? null) + ")").value;
};
const fromGuest = (h) => {
  if (vm.typeof(h) === "function") {
    const keep = h.dup();                                   // not released yet: free it when the cell is redefined
    return (...args) => { const hs = args.map(toGuest); const r = vm.callFunction(keep, vm.undefined, ...hs);
      hs.forEach((a) => a.dispose()); if (r.error) throw new Error(JSON.stringify(vm.dump(r.error)));
      const v = fromGuest(r.value); r.value.dispose(); return v; };
  }
  return vm.dump(h);
};
const interpreted = (definitionSource) => (...inputs) => {
  const fn = vm.evalCode("(" + definitionSource + ")");
  if (fn.error) throw new Error(JSON.stringify(vm.dump(fn.error)));
  const hs = inputs.map(toGuest);
  const r = vm.callFunction(fn.value, vm.undefined, ...hs);
  hs.forEach((a) => a.dispose()); fn.value.dispose();
  if (r.error) throw new Error(JSON.stringify(vm.dump(r.error)));   // becomes the cell's error
  const v = fromGuest(r.value); r.value.dispose(); return v;
};
